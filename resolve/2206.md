 # 벽 부수고 이동하기

### 문제
N×M의 행렬로 표현되는 맵이 있다. 맵에서 0은 이동할 수 있는 곳을 나타내고, 1은 이동할 수 없는 벽이 있는 곳을 나타낸다. 당신은 (1, 1)에서 (N, M)의 위치까지 이동하려 하는데, 이때 최단 경로로 이동하려 한다. 최단경로는 맵에서 가장 적은 개수의 칸을 지나는 경로를 말하는데, 이때 시작하는 칸과 끝나는 칸도 포함해서 센다.

만약에 이동하는 도중에 한 개의 벽을 부수고 이동하는 것이 좀 더 경로가 짧아진다면, 벽을 한 개 까지 부수고 이동하여도 된다.

한 칸에서 이동할 수 있는 칸은 상하좌우로 인접한 칸이다.

맵이 주어졌을 때, 최단 경로를 구해 내는 프로그램을 작성하시오.
### 입력
첫째 줄에 N(1 ≤ N ≤ 1,000), M(1 ≤ M ≤ 1,000)이 주어진다. 다음 N개의 줄에 M개의 숫자로 맵이 주어진다. (1, 1)과 (N, M)은 항상 0이라고 가정하자.
### 출력
첫째 줄에 최단 거리를 출력한다. 불가능할 때는 -1을 출력한다.

from collections import deque

class QUEUE_DICT:
    def __init__(self, x, y, enable_broken_wall):
        self.x = x
        self.y = y
        self.enable_broken_wall = enable_broken_wall
    def get_pos(self):
        return (self.x, self.y)
    def get_enable_broken_wall(self):
        return self.enable_broken_wall
    def put_enable_broken_wall(self, enable_broken_wall):
        self.enable_broken_wall = enable_broken_wall

N,M = map(int,input().split())

graph = []
for _ in range(N):
    tmp_list = list(map(int,input()))
    graph.append(tmp_list)


def bfs():
    queue = deque()
    queue.append(QUEUE_DICT(0,0,True))
    ret_list = {(0,0):{True:1}}
    dir_list = [(-1,0),(1,0),(0,-1),(0,1)]
    while queue:
        q = queue.popleft()
        pos = q.get_pos()
        enable_broken_wall = q.get_enable_broken_wall()
        for d in dir_list:
            x = pos[0] + d[0]
            y = pos[1] + d[1]
            if x<0 or y<0 or x>=N or y>=M:
                continue
            elif not enable_broken_wall and graph[x][y] == 1:
                continue
            ret = ret_list.get((x,y))
            cur_enable_broken_wall = False if graph[x][y] == 1 else True
            cur_enable_broken_wall = cur_enable_broken_wall and enable_broken_wall
            if not ret:
                ret_list[(x,y)] = {cur_enable_broken_wall:ret_list[(pos[0],pos[1])].get(enable_broken_wall)+1}
            elif not ret.get(cur_enable_broken_wall):
                ret_list[(x,y)][cur_enable_broken_wall] = ret_list[(pos[0],pos[1])].get(enable_broken_wall)+1
            else:
                continue
            queue.append(QUEUE_DICT(x,y,cur_enable_broken_wall))
    
    if not ret_list.get((N-1,M-1)):
        print(-1)
    elif ret_list[(N-1,M-1)].get(True) and ret_list[(N-1,M-1)].get(False) :
        print(min(ret_list[(N-1,M-1)].get(True), ret_list[(N-1,M-1)].get(False)))
    elif ret_list[(N-1,M-1)].get(True):
        print(ret_list[(N-1,M-1)].get(True))
    else:
        print(ret_list[(N-1,M-1)].get(False))

from collections import deque
N,M = map(int,input().split())

graph = []
for _ in range(N):
    tmp_list = list(map(int,input()))
    graph.append(tmp_list)


def bfs():
    queue = deque()
    queue.append([0,0,1])
    ret_list = [[[-1]*2 for _ in range(M)] for _ in range(N)]
    ret_list[0][0][1] = 1
    # ret_list = {(0,0):{True:1}}
    dir_list = [(-1,0),(1,0),(0,-1),(0,1)]
    while queue:
        q = queue.popleft()
        pos = (q[0],q[1])
        enable_broken_wall = q[2]
        for d in dir_list:
            x = pos[0] + d[0]
            y = pos[1] + d[1]
            if x<0 or y<0 or x>=N or y>=M:
                continue
            elif enable_broken_wall == 0 and graph[x][y] == 1:
                continue            
            cur_enable_broken_wall = 0 if graph[x][y] == 1 else 1
            if enable_broken_wall == 0:
                cur_enable_broken_wall = 0
            if ret_list[x][y][cur_enable_broken_wall] == -1:
                ret_list[x][y][cur_enable_broken_wall] = ret_list[pos[0]][pos[1]][enable_broken_wall]+1
                queue.append([x,y,cur_enable_broken_wall])
            else:
                continue
            
    check_zero = ret_list[N-1][M-1][0]
    check_one = ret_list[N-1][M-1][1]
    if check_zero == -1 and check_one == -1:
        print(-1)
    elif check_one != -1 and check_zero != -1 :
        print(min(check_one, check_zero))
    elif check_one != -1:
        print(check_one)
    else:
        print(check_zero)
